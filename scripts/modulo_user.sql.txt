-- 1. Primero crear las tablas básicas sin dependencias
CREATE TABLE "user_type" (
    id SERIAL PRIMARY KEY,
    descripcion VARCHAR NOT NULL,
    codigo VARCHAR NOT NULL
);

CREATE TABLE "rol" (
    id SERIAL PRIMARY KEY,
    descripcion VARCHAR NOT NULL
);

CREATE TABLE "modulo" (
    id SERIAL PRIMARY KEY,
    codigo VARCHAR(50) NOT NULL, -- identificador estable (ej: 'usuarios', 'ventas')
    descripcion VARCHAR NOT NULL,       -- nombre legible (ej: 'Gestión de Usuarios')
    status BOOLEAN NOT NULL DEFAULT TRUE,
    cliente_id INTEGER NOT NULL REFERENCES cliente(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP DEFAULT NULL
);

CREATE TABLE "modulo_maestro" (
    id SERIAL PRIMARY KEY,
    codigo VARCHAR(50) NOT NULL, 
    descripcion VARCHAR NOT NULL,      
    status BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE permiso (
    id SERIAL PRIMARY KEY,
    descripcion VARCHAR(150) NOT NULL,
    codigo VARCHAR(100)  NOT NULL,
    modulo_id INTEGER REFERENCES modulo(id) ON DELETE CASCADE,
    cliente_id INTEGER NOT NULL REFERENCES cliente(id) ON DELETE CASCADE,
    es_global BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE permiso_maestro (
    id SERIAL PRIMARY KEY,
    descripcion VARCHAR(150) NOT NULL,
    codigo VARCHAR(100)  NOT NULL,
    es_global BOOLEAN DEFAULT FALSE
);

-- 2. Crear tabla user (depende de user_type)
CREATE TABLE "user" (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR NOT NULL,
    apellido VARCHAR NOT NULL,
    email VARCHAR UNIQUE NOT NULL,
    user_type_id INTEGER REFERENCES "user_type"(id) NOT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP DEFAULT NULL
);

-- 3. Crear tablas que dependen de user
CREATE TABLE "profile" (
    id SERIAL PRIMARY KEY,
    id_user INTEGER REFERENCES "user"(id) NOT NULL,
    dni INTEGER NOT NULL,
    telefono VARCHAR NULL,
    password VARCHAR NOT NULL,
    legajo VARCHAR NOT NULL,
    fecha_ingreso DATE NOT NULL,
    casa_nro VARCHAR DEFAULT NULL,
    barrio VARCHAR DEFAULT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP DEFAULT NULL
);

CREATE TABLE "logs" (
    id SERIAL PRIMARY KEY,
    id_user INTEGER REFERENCES "user"(id) NOT NULL,
    accion VARCHAR NOT NULL,
    descripcion VARCHAR NOT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. Crear tablas de relación many-to-many
CREATE TABLE "user_rol" (
    id SERIAL PRIMARY KEY,
    id_rol INTEGER REFERENCES "rol"(id) NOT NULL,
    id_user INTEGER REFERENCES "user"(id) NOT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
);

CREATE TABLE "modulo_rol" (
    id SERIAL PRIMARY KEY,
    id_rol INTEGER REFERENCES "rol"(id) NOT NULL,
    id_modulo INTEGER REFERENCES "modulo"(id) NOT NULL
);

-- Tabla intermedia rol-permiso
CREATE TABLE rol_permiso (
    id SERIAL PRIMARY KEY,
    rol_id INTEGER NOT NULL REFERENCES rol(id) ON DELETE CASCADE,
    permiso_id INTEGER NOT NULL REFERENCES permiso(id) ON DELETE CASCADE,
    cliente_id INTEGER NOT NULL REFERENCES cliente(id) ON DELETE CASCADE,
    UNIQUE (rol_id, permiso_id, cliente_id)
);

CREATE TABLE cliente (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    cuit VARCHAR(20) UNIQUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
