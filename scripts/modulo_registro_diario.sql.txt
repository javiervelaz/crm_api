-- 1. Primero crear las tablas básicas sin dependencias
CREATE TABLE "sucursal" (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    direccion VARCHAR(255) NOT NULL
);

-- 2. Tablas de catálogos básicos
CREATE TABLE tipo_producto (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    cliente_id INTEGER NOT NULL,
    CONSTRAINT fk_tipo_producto_cliente
        FOREIGN KEY (cliente_id)
        REFERENCES cliente(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
);

CREATE TABLE "medio_pago" (
    id SERIAL PRIMARY KEY,
    descripcion VARCHAR(255) NOT NULL,
    codigo VARCHAR(100) NOT NULL
);

CREATE TABLE categoria_tipo (
  id SERIAL PRIMARY KEY,
  descripcion VARCHAR(20) NOT NULL ,
  codigo VARCHAR(20) NOT NULL
);

-- 3. Asumiendo que la tabla 'user' ya existe, si no debería crearse primero
-- CREATE TABLE "user" (id SERIAL PRIMARY KEY, ...);

-- 4. Crear tabla producto (depende de tipo_producto)
CREATE TABLE "producto" (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    precio_unitario DECIMAL(10, 2) NOT NULL,
    tipo_producto_id INTEGER NOT NULL,
    permite_mitad bool NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    FOREIGN KEY (tipo_producto_id) REFERENCES tipo_producto (id)
);

-- 5. Crear registro_diario (depende de user y sucursal)
CREATE TABLE "registro_diario" (
    id SERIAL PRIMARY KEY,
    fecha DATE NOT NULL,
    usuario_apertura_id INTEGER NOT NULL,
    usuario_cierre_id INTEGER,
    caja_inicial DECIMAL(10, 2) NOT NULL,
    caja_final DECIMAL(10, 2),
    sucursal_id INTEGER NOT NULL,
    ventas_efectivo DECIMAL(10, 2)  NULL,
    ventas_digital DECIMAL(10, 2)  NULL,
    retiros_total DECIMAL(10, 2)  NULL,
    gatos_total DECIMAL(10, 2)  NULL,
    diferencias_caja DECIMAL(10, 2)  NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    caja_cerrada bool NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_apertura_id) REFERENCES "user" (id),
    FOREIGN KEY (usuario_cierre_id) REFERENCES "user" (id),
    FOREIGN KEY (sucursal_id) REFERENCES "sucursal" (id)
);

-- 6. Agregar columnas adicionales a registro_diario
ALTER TABLE registro_diario
  ADD COLUMN ventas_efectivo DECIMAL(10,2),
  ADD COLUMN ventas_digital DECIMAL(10,2),
  ADD COLUMN retiros_total DECIMAL(10,2),
  ADD COLUMN gastos_total DECIMAL(10,2),
  ADD COLUMN diferencias_caja DECIMAL(10,2);

-- 7. Crear tabla pedido (depende de registro_diario, user, sucursal, medio_pago, producto)
CREATE TABLE "pedido" (
    id SERIAL PRIMARY KEY,
    registro_diario_id INTEGER NOT NULL,
    monto_total DECIMAL(10, 2) NOT NULL,
    usuario_id INTEGER NOT NULL,
    sucursal_id INTEGER NOT NULL,
    medio_pago_id INTEGER NOT NULL,
    observacion VARCHAR NULL,
    comanda_nro NUMERIC NULL, 
    user_cliente_id INTEGER NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    pedido_terminado boolean NULL,
    paga_efectivo DECIMAL(10, 2) NULL,
    vuelto_pago_efectivo DECIMAL(10, 2)  NULL,
    monto_adicional numeric(10, 2) NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (registro_diario_id) REFERENCES "registro_diario" (id),
    FOREIGN KEY (usuario_id) REFERENCES "user" (id),
    FOREIGN KEY (sucursal_id) REFERENCES "sucursal" (id),
    FOREIGN KEY (medio_pago_id) REFERENCES "medio_pago" (id)
);

-- 8. Crear tablas dependientes de pedido
CREATE TABLE "pedido_producto" (
    id SERIAL PRIMARY KEY,
    pedido_id INTEGER NOT NULL,
    producto_id INTEGER NOT NULL,
    cantidad INTEGER NOT NULL,
    cantidad_mitad INTEGER NOT NULL,
    observaciones VARCHAR(255) NULL,
    precio_final DECIMAL(10, 2) NULL,
    precio_unitario DECIMAL(10, 2) NOT NULL,
    monto_adicional DECIMAL(10, 2) NULL,
    FOREIGN KEY (pedido_id) REFERENCES "pedido" (id),
    FOREIGN KEY (producto_id) REFERENCES "producto" (id)
);

-- 9. Crear compra_insumo (depende de registro_diario, user, sucursal)
CREATE TABLE "compra_insumo" (
    id SERIAL PRIMARY KEY,
    registro_diario_id INTEGER NOT NULL,
    insumo VARCHAR(100) NOT NULL,
    cantidad INTEGER NOT NULL,
    precio_total DECIMAL(10, 2) NOT NULL,
    usuario_id INTEGER NOT NULL,
    sucursal_id INTEGER NOT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (registro_diario_id) REFERENCES "registro_diario" (id),
    FOREIGN KEY (usuario_id) REFERENCES "user" (id),
    FOREIGN KEY (sucursal_id) REFERENCES "sucursal" (id)
);

-- 10. Crear categoria_salida (depende de categoria_tipo)
CREATE TABLE categoria_salida (
  id SERIAL PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  categoria_tipo_id INTEGER NOT NULL,
  cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
  FOREIGN KEY (categoria_tipo_id) REFERENCES categoria_tipo(id)
);

-- 11. Crear salida_caja (depende de registro_diario, categoria_tipo, user, sucursal)
CREATE TABLE salida_caja (
  id SERIAL PRIMARY KEY,
  registro_diario_id INTEGER NOT NULL,
  categoria_salida_id INTEGER NOT NULL,
  descripcion VARCHAR(255),
  cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
  monto DECIMAL(10, 2) NOT NULL,
  usuario_id INTEGER NOT NULL,
  sucursal_id INTEGER NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (registro_diario_id) REFERENCES registro_diario(id),
  FOREIGN KEY (usuario_id) REFERENCES "user"(id),
  FOREIGN KEY (sucursal_id) REFERENCES sucursal(id)
);

-- 12. Crear tipo enum para conteo_caja
CREATE TYPE tipo_conteo AS ENUM ('billete', 'moneda');

-- 13. Crear conteo_caja (depende de registro_diario)
CREATE TABLE conteo_caja (
  id SERIAL PRIMARY KEY,
  registro_diario_id INTEGER NOT NULL,
  tipo tipo_conteo NOT NULL,
  denominacion NUMERIC(10, 2) NOT NULL,
  cantidad INTEGER NOT NULL,
  FOREIGN KEY (registro_diario_id) REFERENCES registro_diario(id)
);

-- 14. Crear inventario_insumos (depende de sucursal)
CREATE TABLE inventario_insumos (
    id SERIAL PRIMARY KEY,
    sucursal_id INTEGER NOT NULL,
    nombre VARCHAR(255) NOT NULL,
    cantidad DECIMAL(10, 2) NOT NULL,
    cliente_id  INTEGER NOT NULL REFERENCES cliente(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sucursal_id) REFERENCES sucursal (id)
);

-- 15. Crear movimiento_inventario (depende de inventario_insumos y registro_diario)
CREATE TABLE movimiento_inventario (
    id SERIAL PRIMARY KEY,
    inventario_insumos_id INTEGER NOT NULL,
    registro_diario_id INTEGER NOT NULL,
    tipo_movimiento VARCHAR(50) NOT NULL,
    cantidad DECIMAL(10, 2) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (inventario_insumos_id) REFERENCES inventario_insumos (id),
    FOREIGN KEY (registro_diario_id) REFERENCES registro_diario (id)
);

-- 16. Crear producto_img (depende de producto)
CREATE TABLE "producto_img" (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion VARCHAR(255) NOT NULL,
    producto_id INTEGER NOT NULL,
    FOREIGN KEY (producto_id) REFERENCES producto (id)
);

-- 17. Crear tablas de WhatsApp (dependen de user y pedido)
CREATE TABLE whatsapp_conversation (
    id SERIAL PRIMARY KEY,
    cliente_id INTEGER NULL,
    telefono VARCHAR(20) NOT NULL,
    estado VARCHAR(20) NOT NULL DEFAULT 'menu',
    state VARCHAR(20) NULL,
    cart JSONB NOT NULL DEFAULT '{}',
    sucursal_id INTEGER NULL,
    last_message_at TIMESTAMPTZ DEFAULT NOW(),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (cliente_id) REFERENCES "cliente" (id)
);

CREATE INDEX IF NOT EXISTS idx_whatsapp_conversation_cliente_telefono
  ON whatsapp_conversation (cliente_id, telefono);

CREATE INDEX IF NOT EXISTS idx_whatsapp_conversation_cliente_state
  ON whatsapp_conversation (cliente_id, estado);

CREATE TABLE IF NOT EXISTS whatsapp_message (
  id BIGSERIAL PRIMARY KEY,
  conversation_id BIGINT NOT NULL REFERENCES whatsapp_conversation(id),
  wa_message_id TEXT,                         -- id de WhatsApp
  direction VARCHAR(10) NOT NULL,             -- 'in' | 'out'
  body TEXT,                                  -- texto plano
  raw JSONB,                                  -- payload completo
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_whatsapp_message_conversation
  ON whatsapp_message (conversation_id);
CREATE UNIQUE INDEX IF NOT EXISTS uq_whatsapp_message_waid_in
  ON whatsapp_message (wa_message_id, direction)
  WHERE direction = 'in' AND wa_message_id IS NOT NULL;
